cmake_minimum_required(VERSION 3.20)
project(tt VERSION 0.1.0 LANGUAGES CXX)
if(NOT UNIX OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "The prototype currently supports GCC/Clang on POSIX hosts")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TT_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(TT_BUILD_BENCHMARKS "Build the in-process compiler benchmark" ON)
include(CTest)
function(tt_options target)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wconversion -Wshadow
    -Wno-misleading-indentation -Wno-missing-field-initializers)
  if(TT_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
add_executable(tt src/main.cpp)
tt_options(tt)
install(TARGETS tt RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  find_package(Python3 3.9 COMPONENTS Interpreter REQUIRED)
  add_executable(tt-kernel tests/kernel.cpp)
  tt_options(tt-kernel)
  add_test(NAME conformance COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:tt>)
  add_test(NAME kernel COMMAND tt-kernel)
  add_test(NAME local-workflow COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_SOURCE_DIR}/tests/test_dev.py)
endif()
if(TT_BUILD_BENCHMARKS)
  add_executable(tt-bench benchmarks/bench.cpp)
  tt_options(tt-bench)
endif()
